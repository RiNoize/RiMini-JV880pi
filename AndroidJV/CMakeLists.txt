cmake_minimum_required(VERSION 3.22)
project(RiJV880Android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Android arm64-v8a toolchain required")
endif()
add_subdirectory("${JUCE_ROOT}/modules/juce_audio_devices/native/oboe" oboe)
add_library(cpufeatures STATIC "${ANDROID_NDK}/sources/android/cpufeatures/cpu-features.c")
file(GLOB_RECURSE ENGINE CONFIGURE_DEPENDS "${VIRTUALJV_ROOT}/Source/*.cpp" "${VIRTUALJV_ROOT}/Source/emulator/resample/*.c")
set(SOURCES Main.cpp ${ENGINE})
foreach(module audio_basics audio_devices audio_formats audio_processors audio_processors_headless audio_processors_headless_ara audio_processors_headless_lv2_libs audio_utils core core_CompilationTime data_structures events graphics graphics_Harfbuzz gui_basics gui_extra midi_ci)
  string(REGEX REPLACE "_(CompilationTime|Harfbuzz|ara|lv2_libs)$" "" folder "${module}")
  list(APPEND SOURCES "${JUCE_ROOT}/modules/juce_${folder}/juce_${module}.cpp")
endforeach()
foreach(part RANGE 2 5)
  list(APPEND SOURCES "${JUCE_ROOT}/modules/juce_gui_basics/juce_gui_basics_${part}.cpp")
endforeach()
list(APPEND SOURCES "${JUCE_ROOT}/modules/juce_graphics/juce_graphics_Sheenbidi.c")
add_library(juce_jni SHARED ${SOURCES})
target_include_directories(juce_jni PRIVATE . "${VIRTUALJV_ROOT}/Source" "${JUCE_ROOT}/modules" "${ANDROID_NDK}/sources/android/cpufeatures")
target_compile_options(juce_jni PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/BuildConfig.h" -O2)
set_source_files_properties(${ENGINE} PROPERTIES COMPILE_OPTIONS "-O3")
target_compile_definitions(juce_jni PRIVATE JUCE_ANDROID=1 JUCE_ANDROID_API_VERSION=29 JUCE_STANDALONE_APPLICATION=1 JUCE_USE_ANDROID_OBOE=1 JUCE_PUSH_NOTIFICATIONS=0 JUCE_ANDROID_GL_ES_VERSION_3_0=1 NDEBUG=1)
target_link_options(juce_jni PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined)
target_link_libraries(juce_jni PRIVATE oboe cpufeatures log android EGL GLESv2 GLESv1_CM dl atomic m)
